cmake_minimum_required(VERSION 3.24)
project(WarfarerSteamCloud LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build for Windows x64")
endif()
include(FetchContent)
# Match REFramework's Lua ABI; do not link a second Steam API DLL.
FetchContent_Declare(lua URL https://www.lua.org/ftp/lua-5.4.3.tar.gz
  URL_HASH SHA256=f8612276169e3bfcbcfb8f226195bfc6e466fe13042f1076cbde92b7ec96bbfb)
FetchContent_MakeAvailable(lua)
set(REF_COMMIT d052e8fd3fd41d0a1a33394fcf6e6c19be39b1ee)
file(DOWNLOAD https://raw.githubusercontent.com/praydog/REFramework/${REF_COMMIT}/include/reframework/API.h
  ${CMAKE_CURRENT_BINARY_DIR}/reframework/API.h STATUS download_status TLS_VERIFY ON)
list(GET download_status 0 download_code)
if(NOT download_code EQUAL 0)
  message(FATAL_ERROR "Cannot download REFramework API.h: ${download_status}")
endif()
file(GLOB LUA_SOURCES ${lua_SOURCE_DIR}/src/*.c)
list(FILTER LUA_SOURCES EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(lua STATIC ${LUA_SOURCES})
target_include_directories(lua PUBLIC ${lua_SOURCE_DIR}/src)
add_executable(warfarer_lua ${lua_SOURCE_DIR}/src/lua.c)
target_link_libraries(warfarer_lua PRIVATE lua)
add_library(WarfarerSteamCloud SHARED plugin.cpp)
set_target_properties(WarfarerSteamCloud PROPERTIES PREFIX "")
if(MINGW)
  target_link_options(WarfarerSteamCloud PRIVATE -static-libgcc -static-libstdc++ -static)
endif()
target_compile_features(WarfarerSteamCloud PRIVATE cxx_std_17)
target_include_directories(WarfarerSteamCloud PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(WarfarerSteamCloud PRIVATE lua)
install(TARGETS WarfarerSteamCloud RUNTIME DESTINATION reframework/plugins)
install(FILES ${lua_SOURCE_DIR}/doc/readme.html DESTINATION reframework/plugins/WarfarerSteamCloud-licenses RENAME Lua.html)
add_executable(cloud_smoke ../../tests/cloud-smoke.cpp)
target_include_directories(cloud_smoke PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(cloud_smoke PRIVATE lua)
